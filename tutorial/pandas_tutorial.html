1. Introduction

Pandas is the most popular Python library for data analysis and manipulation. It provides two main data structures:

  • Series — 1-dimensional labeled array
  • DataFrame — 2-dimensional table (like an Excel sheet or SQL table)

Pandas is built on top of NumPy and is excellent for reading CSV/Excel files, cleaning data, filtering, aggregating, and preparing data for analysis or machine learning.

2. Installation & Import

# Install (once)
pip install pandas openpyxl   # openpyxl needed for Excel (.xlsx)

# Import convention
import pandas as pd
import numpy as np
Always use the alias pd. Almost every tutorial and Stack Overflow answer expects it.

3. Reading CSV & Excel Files

Read CSV

# Basic
df = pd.read_csv("data.csv")

# Common options
df = pd.read_csv(
    "data.csv",
    sep=",",                 # delimiter (default is comma)
    header=0,               # row number to use as column names
    index_col=0,            # use first column as index
    usecols=["name", "age", "salary"],  # only these columns
    nrows=1000,             # read only first 1000 rows
    na_values=["NA", "?", ""],  # treat these as missing
    parse_dates=["date"],   # parse date column
    encoding="utf-8"
)

Read Excel

# Single sheet
df = pd.read_excel("data.xlsx")

# Specific sheet
df = pd.read_excel("data.xlsx", sheet_name="Sales")

# Multiple sheets → dict of DataFrames
dfs = pd.read_excel("data.xlsx", sheet_name=["Sales", "Customers"])
# or sheet_name=None to load all sheets

# Useful options
df = pd.read_excel(
    "data.xlsx",
    sheet_name=0,           # first sheet (or name)
    header=0,
    usecols="A:D",         # columns A to D
    skiprows=2,            # skip first 2 rows
    nrows=500
)
For .xlsx files you need the openpyxl engine (installed above).
For old .xls files use engine="xlrd" (and install xlrd).

Other common readers

pd.read_json("data.json")
pd.read_parquet("data.parquet")   # fast & efficient
pd.read_sql("SELECT * FROM table", connection)
pd.read_clipboard()               # paste from Excel / Google Sheets

5. Selecting Columns & Rows

Select columns

# Single column → Series
df["name"]
df.name                # also works if column name is a valid identifier

# Multiple columns → DataFrame
df[["name", "age", "salary"]]

Select rows by position (iloc) or label (loc)

# iloc → integer position
df.iloc[0]            # first row
df.iloc[0:5]          # first 5 rows
df.iloc[0:5, 0:3]      # first 5 rows, first 3 columns
df.iloc[:, [0, 2]]     # all rows, columns 0 and 2

# loc → label based
df.loc[0]             # row with index label 0
df.loc[0:4, ["name", "salary"]]
df.loc[df["age"] > 30, "name"]  # names of people older than 30
iloc uses integer positions (0-based).
loc uses the actual index labels (can be numbers, strings, dates…).

6. Filtering (Boolean Indexing)

This is one of the most common operations in Pandas.

# Simple condition
df[df["age"] > 30]

# Multiple conditions (use & | ~  and parentheses!)
df[(df["age"] > 30) & (df["salary"] > 50000)]
df[(df["department"] == "Sales") | (df["department"] == "Marketing")]

# isin — cleaner for multiple values
df[df["department"].isin(["Sales", "Marketing", "HR"])]

# String methods
df[df["name"].str.startswith("A")]
df[df["email"].str.contains("@gmail.com", na=False)]
df[df["city"].str.contains("New|Los", case=False, na=False)]

# Between
df[df["age"].between(25, 40)]

# Query method (nice for complex filters)
df.query("age > 30 and salary > 50000")
df.query("department == 'Sales' and age >= 25")
Always put each condition in parentheses when combining with & or |.

7. Common Operations

Add / modify columns

df["bonus"] = df["salary"] * 0.1
df["full_name"] = df["first"] + " " + df["last"]
df["age_group"] = np.where(df["age"] >= 40, "Senior", "Junior")

# Using assign (returns new DataFrame)
df = df.assign(
    tax = df["salary"] * 0.2,
    net  = lambda x: x["salary"] - x["tax"]
)

Rename columns

df = df.rename(columns={"old_name": "new_name", "sal": "salary"})
df.columns = [c.lower().strip() for c in df.columns]  # normalize

Drop columns / rows

df = df.drop(columns=["temp", "notes"])
df = df.drop(index=[0, 5, 10])
df = df.dropna()                     # drop rows with any NaN
df = df.drop_duplicates()
df = df.drop_duplicates(subset=["email"])

Sorting

df.sort_values("salary", ascending=False)
df.sort_values(["department", "salary"], ascending=[True, False])
df.sort_index()

Apply custom functions

df["name_len"] = df["name"].apply(len)
df["salary_k"] = df["salary"].apply(lambda x: f"{x/1000:.1f}k")

# Apply to whole row
df["desc"] = df.apply(
    lambda row: f"{row['name']} works in {row['department']}",
    axis=1
)

8. Handling Missing Data

df.isna()                    # True where missing
df.isna().sum()              # count missing per column
df.notna()

# Drop
df.dropna()                  # drop any row with NaN
df.dropna(subset=["salary"]) # only if salary is missing
df.dropna(axis=1)            # drop columns with any NaN

# Fill
df["age"] = df["age"].fillna(df["age"].mean())
df["department"] = df["department"].fillna("Unknown")
df.fillna(method="ffill")     # forward fill
df.fillna(method="bfill")     # backward fill

# Replace specific values
df.replace({"NA": np.nan, "?": np.nan}, inplace=True)

9. GroupBy & Aggregation

# Basic
df.groupby("department")["salary"].mean()
df.groupby("department")["salary"].agg(["mean", "min", "max", "count"])

# Multiple columns
df.groupby("department").agg({
    "salary": "mean",
    "age": ["min", "max"],
    "name": "count"
})

# Named aggregations (clean)
df.groupby("department").agg(
    avg_salary=("salary", "mean"),
    max_age=("age", "max"),
    employees=("name", "count")
)

# Transform (keeps original shape)
df["dept_avg_salary"] = df.groupby("department")["salary"].transform("mean")

10. Merge, Join & Concat

# Concatenate (stack rows or columns)
pd.concat([df1, df2], axis=0, ignore_index=True)  # stack vertically
pd.concat([df1, df2], axis=1)                    # side by side

# Merge (SQL-style joins)
pd.merge(left, right, on="id", how="inner")
pd.merge(left, right, on="id", how="left")
pd.merge(left, right, left_on="emp_id", right_on="id", how="outer")

# Join on index
left.join(right, how="left")
how=Meaning
innerOnly matching keys (default)
leftAll from left + matches from right
rightAll from right + matches from left
outerAll keys from both

12. Worked Examples

Example 1 — Clean & filter employee data

df = pd.read_csv("employees.csv")

# Clean
df["salary"] = pd.to_numeric(df["salary"], errors="coerce")
df = df.dropna(subset=["salary", "department"])
df["name"] = df["name"].str.title().str.strip()

# Filter high earners in Sales or Engineering
result = df[
    (df["salary"] > 70000) &
    (df["department"].isin(["Sales", "Engineering"]))
].sort_values("salary", ascending=False)

print(result[["name", "department", "salary"]].head(10))

Example 2 — Summary by department

summary = (
    df.groupby("department")
      .agg(
          employees=("name", "count"),
          avg_salary=("salary", "mean"),
          median_salary=("salary", "median"),
          max_salary=("salary", "max")
      )
      .round(0)
      .sort_values("avg_salary", ascending=False)
)

print(summary)
summary.to_excel("department_summary.xlsx")

Example 3 — Read Excel, process, write multiple sheets

# Read
sales = pd.read_excel("raw_data.xlsx", sheet_name="Sales")
customers = pd.read_excel("raw_data.xlsx", sheet_name="Customers")

# Merge
merged = pd.merge(sales, customers, on="customer_id", how="left")

# Add month column
merged["month"] = pd.to_datetime(merged["date"]).dt.to_period("M")

# Monthly totals
monthly = merged.groupby("month")["amount"].sum().reset_index()

# Write report
with pd.ExcelWriter("sales_report.xlsx") as writer:
    merged.to_excel(writer, sheet_name="All Sales", index=False)
    monthly.to_excel(writer, sheet_name="Monthly", index=False)

Example 4 — Quick filtering patterns

# Top 10 highest salaries
df.nlargest(10, "salary")

# Rows where salary is above department average
df["dept_avg"] = df.groupby("department")["salary"].transform("mean")
above_avg = df[df["salary"] > df["dept_avg"]]

# Filter with query + variable
min_sal = 60000
df.query("salary >= @min_sal and department == 'Engineering'")
